# 合并片段，不是完整配置，也不是插件；不要覆盖整份主配置。
# 在主配置已有的 [General] 内逐项更新，避免重复键。

[General]
ip-mode = v4-only
hijack-dns = *:53
disable-stun = true
udp-fallback-mode = REJECT

# 加密 DNS 示例：在 Loon DNS 页面选择/添加，并关闭普通 DNS 回落。
# 使用 IP 形式减少解析 DoH 服务器域名的引导依赖。
# 此设置仅配置加密解析，尚不证明 Loon 自身 DNS 查询走代理！
doh-server = https://1.1.1.1/dns-query

# 不在这里伪造 DNS 代理参数。需要在目标版本中核验 DNS 的实际出站路径。
